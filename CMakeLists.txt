cmake_minimum_required(VERSION 3.24)
project(toker_cs2 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(FetchContent)
set(TOKER_DEPS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor" CACHE PATH "Directory containing imgui, minhook, json; empty to fetch")
if(TOKER_DEPS_DIR)
  set(FETCHCONTENT_SOURCE_DIR_JSON "${TOKER_DEPS_DIR}/json")
  set(FETCHCONTENT_SOURCE_DIR_IMGUI "${TOKER_DEPS_DIR}/imgui")
  set(FETCHCONTENT_SOURCE_DIR_MINHOOK "${TOKER_DEPS_DIR}/minhook")
endif()
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json)
add_library(toker_core STATIC src/config.cpp)
target_include_directories(toker_core PUBLIC include)
target_link_libraries(toker_core PUBLIC nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(toker_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(toker_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(toker_tests tests/core_tests.cpp)
  target_link_libraries(toker_tests PRIVATE toker_core)
  add_test(NAME core COMMAND toker_tests)
  set_tests_properties(core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Select an x64 toolchain: CS2 is 64-bit")
  endif()
  FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.8 GIT_SHALLOW TRUE)
  FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG v1.3.4 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(imgui minhook)
  add_library(toker_imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
  target_include_directories(toker_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_compile_definitions(toker_imgui PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(toker_imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
  add_library(toker_ui STATIC src/ui.cpp src/demo_scene.cpp)
  target_link_libraries(toker_ui PUBLIC toker_core toker_imgui)
  add_library(toker_cs2 SHARED src/dllmain.cpp src/hook.cpp src/game.cpp)
  target_link_libraries(toker_cs2 PRIVATE toker_ui minhook)
  target_compile_definitions(toker_cs2 PRIVATE UNICODE _UNICODE)
  set_target_properties(toker_cs2 PROPERTIES PREFIX "")
  add_executable(toker_preview WIN32 src/preview.cpp)
  target_link_libraries(toker_preview PRIVATE toker_ui)
  target_compile_definitions(toker_preview PRIVATE UNICODE _UNICODE)
  foreach(target toker_cs2 toker_preview)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${CMAKE_CURRENT_SOURCE_DIR}/config" "$<TARGET_FILE_DIR:${target}>/config")
  endforeach()
endif()
