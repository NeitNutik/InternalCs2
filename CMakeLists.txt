cmake_minimum_required(VERSION 3.24)
project(toker_standoff2 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_subdirectory(vendor/json)
add_library(toker_core STATIC src/config.cpp src/layout.cpp)
target_include_directories(toker_core PUBLIC include)
target_link_libraries(toker_core PUBLIC nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(toker_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(toker_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(toker_tests tests/core_tests.cpp)
  target_link_libraries(toker_tests PRIVATE toker_core)
  add_test(NAME core COMMAND toker_tests)
  set_tests_properties(core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build the Windows reader as x64; guest pointers may be 32 or 64 bit")
  endif()
  set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/imgui")
  add_library(toker_imgui STATIC
    ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp
    ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/backends/imgui_impl_win32.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp)
  target_include_directories(toker_imgui PUBLIC ${IMGUI_DIR} ${IMGUI_DIR}/backends)
  target_compile_definitions(toker_imgui PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(toker_imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
  add_library(toker_ui STATIC src/ui.cpp src/demo_scene.cpp)
  target_link_libraries(toker_ui PUBLIC toker_core toker_imgui)
  add_executable(toker_standoff2 WIN32 src/overlay.cpp src/game.cpp)
  target_link_libraries(toker_standoff2 PRIVATE toker_ui)
  add_executable(toker_preview WIN32 src/preview.cpp)
  target_link_libraries(toker_preview PRIVATE toker_ui)
  foreach(target toker_standoff2 toker_preview)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/config" "$<TARGET_FILE_DIR:${target}>/config")
  endforeach()
endif()
